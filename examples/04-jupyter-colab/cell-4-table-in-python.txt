# The regression table, as a pandas DataFrame (Python)
import pandas as pd
from pystata import stata
from sfi import Matrix

stata.run("regress price mpg weight", quietly=True)
b = Matrix.get("e(b)")[0]
V = Matrix.get("e(V)")
table = pd.DataFrame(
    {"Coefficient": b, "Std. err.": [V[i][i] ** 0.5 for i in range(len(b))]},
    index=["Mileage (mpg)", "Weight (lbs.)", "Intercept"],
).round(2)

# the same table, as text, for the Word and PDF documents
rows = [["", *table.columns]] + table.reset_index().astype(str).values.tolist()
